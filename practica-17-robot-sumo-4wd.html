Nombre de la práctica

Programación de comportamiento básico para robot sumo autónomo

Propósito

El alumno comprenderá cómo funciona un robot sumo autónomo capaz de detectar el borde del dojo, evitar salirse del área de combate, buscar al oponente girando, atacar cuando detecta un objeto al frente y calibrarse automáticamente según el color del piso.

1. Materiales necesarios

  • 1 Arduino UNO o compatible
  • 1 módulo L298N
  • 1 chasis 4WD
  • 4 motores DC
  • 3 sensores TCRT5000 analógicos
  • 1 sensor ultrasónico HC-SR04
  • Batería para motores
  • Cables Dupont
  • Base o dojo de prueba
  • Computadora con Arduino IDE

4. Explicación por bloques

A) Pines del L298N

const int IN1 = 7;
const int IN2 = 4;
const int IN3 = 3;
const int IN4 = 2;
const int ENA = 5;
const int ENB = 6;

Estos pines controlan los motores. IN1, IN2, IN3, IN4 controlan la dirección. ENA y ENB controlan la velocidad mediante PWM.

B) Pines de los sensores TCRT5000

const int IR_LEFT   = A0;
const int IR_CENTER = A1;
const int IR_RIGHT  = A4;

Estos sensores detectan el cambio de color entre el piso seguro y el borde del dojo.

C) Pines del ultrasónico

const int TRIG = A2;
const int ECHO = A3;

El ultrasónico mide la distancia al oponente.

D) Variable para invertir motores

bool DRIVE_INVERTED = false;

Si el robot avanza hacia atrás cuando debería avanzar, se puede cambiar a:

bool DRIVE_INVERTED = true;

Esto evita tener que cambiar físicamente los cables de los motores.

E) Velocidades

int SPEED_ATTACK = 170;
int SPEED_SEARCH = 115;
int SPEED_ESCAPE = 245;
int SPEED_SPIN   = 110;
VariableFunción
SPEED_ATTACKVelocidad al atacar
SPEED_SEARCHVelocidad al buscar
SPEED_ESCAPEVelocidad para escapar del borde
SPEED_SPINVelocidad del giro inicial

F) Detección del borde

int FLOOR_TYPE_LIMIT = 500;
int EDGE_MARGIN = 18;

FLOOR_TYPE_LIMIT ayuda a decidir si el piso es claro u oscuro y EDGE_MARGIN define qué tanto debe cambiar la lectura para considerar que hay borde.

G) Distancia de ataque

int ENEMY_DISTANCE = 60;

Si el ultrasónico detecta un objeto a 60 cm o menos, el robot ataca.

5. Funciones de movimiento

Detener motores

void motorStop()

Sirve para apagar ambos motores.

Avanzar

void motorForward(int s)

Hace que el robot avance con velocidad s.

Retroceder

void motorBackward(int s)

Hace que el robot retroceda.

Girar a la izquierda

void turnLeft(int s)

Hace que un lado avance y el otro retroceda, permitiendo que el robot gire sobre su eje.

Girar a la derecha

void turnRight(int s)

Hace el giro contrario al anterior.

6. Calibración del piso

El código primero mide el valor del piso seguro con los tres sensores:

baseL = avgRead(IR_LEFT);
baseC = avgRead(IR_CENTER);
baseR = avgRead(IR_RIGHT);

Después calcula un promedio:

int avg = (baseL + baseC + baseR) / 3;

Y decide si el piso es claro u oscuro:

floorIsLight = avg < FLOOR_TYPE_LIMIT;

7. Detección del borde

bool detectEdge(int value, int base)

Si el piso es claro:

return value > base + EDGE_MARGIN;

Si el piso es oscuro:

return value < base - EDGE_MARGIN;

8. Detección del oponente

long getDistanceCM()

Si no detecta nada, devuelve:

return 999;

Si detecta un objeto, calcula la distancia en centímetros:

return duration / 58;

long getDistanceCM() {
  digitalWrite(TRIG, LOW); delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  long duration = pulseIn(ECHO, HIGH, 22000);
  if (duration == 0) return 999;
  return duration / 58;
}

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
}

void loop() {
  int rawL = analogRead(IR_LEFT);
  int rawC = analogRead(IR_CENTER);
  int rawR = analogRead(IR_RIGHT);
  long d = getDistanceCM();

  Serial.print("RAW L="); Serial.print(rawL);
  Serial.print(" C="); Serial.print(rawC);
  Serial.print(" R="); Serial.print(rawR);
  Serial.print(" | D="); Serial.println(d);
  delay(200);
}

Práctica 4: Calibración del piso

const int IR_LEFT = A0, IR_CENTER = A1, IR_RIGHT = A4;
int FLOOR_TYPE_LIMIT = 500;
int baseL = 0, baseC = 0, baseR = 0;
bool floorIsLight = true;

int avgRead(int pin) {
  long sum = 0;
  for (int i = 0; i < 25; i++) {
    sum += analogRead(pin);
    delay(2);
  }
  return sum / 25;
}

void setup() {
  Serial.begin(9600);
  delay(800);
  baseL = avgRead(IR_LEFT);
  baseC = avgRead(IR_CENTER);
  baseR = avgRead(IR_RIGHT);
  int avg = (baseL + baseC + baseR) / 3;
  floorIsLight = avg < FLOOR_TYPE_LIMIT;
  Serial.print("BASE L="); Serial.print(baseL);
  Serial.print(" C="); Serial.print(baseC);
  Serial.print(" R="); Serial.print(baseR);
  Serial.print(" AVG="); Serial.println(avg);
  Serial.println(floorIsLight ? "Piso=CLARO" : "Piso=OSCURO");
}

void loop() {}

Práctica 11: Arranque reglamentario

const int IN1 = 7, IN2 = 4, IN3 = 3, IN4 = 2, ENA = 5, ENB = 6;
int SPEED_SPIN = 110;
unsigned long START_DELAY_TIME = 800;
unsigned long START_SPIN_TIME = 350;
unsigned long startTime = 0;
bool startDone = false;

void motorStop() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);
  analogWrite(ENA, 0); analogWrite(ENB, 0);
}

void turnLeft(int s) {
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
  analogWrite(ENA, s); analogWrite(ENB, s);
}

void setup() {
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  pinMode(ENA, OUTPUT); pinMode(ENB, OUTPUT);
  startTime = millis();
}

void loop() {
  if (!startDone) {
    if (millis() - startTime < START_DELAY_TIME) {
      motorStop();
      return;
    }
    if (millis() - startTime < START_DELAY_TIME + START_SPIN_TIME) {
      turnLeft(SPEED_SPIN);
      return;
    }
    motorStop();
    startDone = true;
  }
}

  } else {
    digitalWrite(IN1, LOW);  digitalWrite(IN2, HIGH);
    digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
  }
  analogWrite(ENA, s); analogWrite(ENB, s);
}

void turnRight(int s) {
  if (!DRIVE_INVERTED) {
    digitalWrite(IN1, LOW);  digitalWrite(IN2, HIGH);
    digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
  } else {
    digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
    digitalWrite(IN3, LOW);  digitalWrite(IN4, HIGH);
  }
  analogWrite(ENA, s); analogWrite(ENB, s);
}

// ================= CALIBRACION =================
int avgRead(int pin) {
  long sum = 0;
  for (int i = 0; i < 25; i++) {
    sum += analogRead(pin);
    delay(2);
  }
  return sum / 25;
}

void calibrateFloor() {
  motorStop();
  Serial.println("Calibrando piso seguro... NO mover");

  baseL = avgRead(IR_LEFT);
  baseC = avgRead(IR_CENTER);
  baseR = avgRead(IR_RIGHT);

  int avg = (baseL + baseC + baseR) / 3;
  floorIsLight = avg < FLOOR_TYPE_LIMIT;

  Serial.print("BASE L=");
  Serial.print(baseL);
  Serial.print(" C=");
  Serial.print(baseC);
  Serial.print(" R=");
  Serial.print(baseR);
  Serial.print(" AVG=");
  Serial.print(avg);
  Serial.print(" Piso=");
  Serial.println(floorIsLight ? "CLARO" : "OSCURO");
}

bool detectEdge(int value, int base) {
  if (floorIsLight) {
    return value > base + EDGE_MARGIN;
  } else {
    return value < base - EDGE_MARGIN;
  }
}

// ================= ULTRASONIDO =================
long getDistanceCM() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH, 22000);
  if (duration == 0) return 999;

  return duration / 58;
}

  Serial.print("RAW L=");

// ================= SETUP =================
void setup() {
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  pinMode(ENA, OUTPUT); pinMode(ENB, OUTPUT);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  Serial.begin(9600);
  motorStop();
}

// ================= LOOP =================
void loop() {
  if (!initDone) {
    motorStop();
    delay(350);

    calibrateFloor();

    motorStop();
    delay(200);

    startTime = millis();
    lastSearchSwap = millis();
    initDone = true;

    Serial.println("Listo. Arrancar sobre zona segura.");
    return;
  }

  int rawL = analogRead(IR_LEFT);
  int rawC = analogRead(IR_CENTER);
  int rawR = analogRead(IR_RIGHT);

  bool edgeL = detectEdge(rawL, baseL);
  bool edgeC = detectEdge(rawC, baseC);
  bool edgeR = detectEdge(rawR, baseR);

  long distance = getDistanceCM();

  printDebug(rawL, rawC, rawR, edgeL, edgeC, edgeR, distance);

  if (!startDone) {
    if (edgeL || edgeC || edgeR) {
      motorBackward(SPEED_ESCAPE);
      delay(600);
      motorStop();
      delay(100);
      startDone = true;
      return;
    }

    if (millis() - startTime < START_DELAY_TIME) {
      motorStop();
      return;
    }

    turnLeft(SPEED_SPIN);

    if (millis() - startTime >= START_DELAY_TIME + START_SPIN_TIME) {
      motorStop();
      delay(100);
      startDone = true;
      lastSearchSwap = millis();
    }

    return;
  }

  if (edgeC) {
    motorBackward(SPEED_ESCAPE);
    delay(ESCAPE_BACK_CENTER_TIME);
    turnRight(SPEED_ESCAPE);
    delay(ESCAPE_TURN_TIME);
    motorStop();
    delay(50);
    return;
  }

  if (edgeL) {
    motorBackward(SPEED_ESCAPE);
    delay(ESCAPE_BACK_SIDE_TIME);
    turnRight(SPEED_ESCAPE);
    delay(ESCAPE_TURN_TIME);
    motorStop();
    delay(50);
    return;
  }

  if (edgeR) {
    motorBackward(SPEED_ESCAPE);
    delay(ESCAPE_BACK_SIDE_TIME);
    turnLeft(SPEED_ESCAPE);
    delay(ESCAPE_TURN_TIME);
    motorStop();
    delay(50);
    return;
  }

  if (distance <= ENEMY_DISTANCE) {
    motorForward(SPEED_ATTACK);
    delay(40);
    return;
  }

  if (millis() - lastSearchSwap >= SEARCH_SWAP_TIME) {
    searchLeft = !searchLeft;
    lastSearchSwap = millis();
  }

  if (searchLeft) {
    turnLeft(SPEED_SEARCH);
  } else {
    turnRight(SPEED_SEARCH);
  }

  delay(10);
}
